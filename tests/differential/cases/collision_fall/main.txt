# collision_fall
# `env a|v` lines carry exactly one field; trailing tokens are a hard error.
# ItrEffect: Normal=0 Sharp=1 Fire=2 Ice=3 Through=4 None=5 MFire1=20 MFire2=21 FireExplosion=22 Explosion=23 Ice2=30 Ignore=10000
# WeaponEnum: Heavy=2
# fly threshold is DEFAULT_FALL_VALUE_CRITICAL = 140 - DEFAULT_FALL_VALUE_DIZZY = 100
env dataset o 1 min_arest n 0
env acube o 6 left n 1 right n 2 bottom n 3 top n 4 near n 5 far n 6
env bcube o 6 left n 7 right n 8 bottom n 9 top n 10 near n 11 far n 12
env a face n 1
env v face n 1
env v hp n 50
env v hp_r n 40
env v fall n 77
env v state n 0
env v fire a 2 s "f0" s "f1"
env v crit o 2 1 a 2 s "c1a" s "c1b" -1 a 2 s "cm1" s "cm2"
env v holding o 1 base_type n 3
env velx 0
env vely 0
env velz 0
# default branch, zero x velocity -> direction 1
env itr o 2 effect n 0 fall n 0
run fall
# positive x velocity -> direction 1
env velx 10
run fall
# negative x velocity -> direction -1
env velx -10
run fall
# victim facing flips the sign back
env v face n -1
run fall
# zero over zero is NaN, which is not >= 0 -> direction -1
env velx 0
env v face n 0
run fall
# positive over zero is +infinity -> direction 1
env velx 5
run fall
# negative over zero is -infinity -> direction -1
env velx -5
run fall
env velx 0
env v face n 1
# a missing critical_hit list short circuits
env v crit u
run fall
env v crit o 2 1 a 2 s "c1a" s "c1b" -1 a 2 s "cm1" s "cm2"
# an empty direction list yields undefined
env v crit o 2 1 a 0 -1 a 2 s "cm1" s "cm2"
run fall
env v crit o 2 1 a 2 s "c1a" s "c1b" -1 a 2 s "cm1" s "cm2"
# sharp plus critical -> critical bleed
env itr o 2 effect n 1 fall n 100
run fall
# sharp without critical -> bleed fall
env itr o 2 effect n 1 fall n 99
run fall
# non sharp with critical -> critical hit
env itr o 2 effect n 0 fall n 100
run fall
# the critical list entry may be a keyed object itself
env v crit o 2 1 o 1 0 s "obj0" -1 a 2 s "cm1" s "cm2"
env itr o 2 effect n 0 fall n 0
run fall
env v crit o 2 1 a 2 s "c1a" s "c1b" -1 a 2 s "cm1" s "cm2"
# the critical list may be an array of lists
env v crit a 2 a 2 s "c1a" s "c1b" a 2 s "cm1" s "cm2"
env itr o 2 effect n 0 fall n 0
run fall
env v crit o 2 1 a 2 s "c1a" s "c1b" -1 a 2 s "cm1" s "cm2"
# non sharp without critical -> hit fall
env itr o 2 effect n 0 fall n 99
run fall
# a non fighter victim skips the bleed kinds
env v type n 0
env itr o 2 effect n 1 fall n 100
run fall
env v type n 8
# fall as a numeric string still counts as critical
env itr o 2 effect n 0 fall s "150"
run fall
# y and z are carried into the velocity write
env velx 3
env vely 5
env velz 7
env itr o 2 effect n 0 fall n 0
run fall
env vely 0
env velz 0
# fire plus a held heavy weapon drops it first
env v holding o 1 base_type n 2
env itr o 2 effect n 2 fall n 0
run fall
# fire plus a light holding keeps it
env v holding o 1 base_type n 3
run fall
# fire with no holding at all
env v holding u
run fall
# mfire2 behaves like fire
env v holding o 1 base_type n 3
env itr o 2 effect n 21 fall n 0
run fall
# the fire turn is mirrored against the attacker facing
env a face n -1
run fall
env a face n 1
# a non one facing is turned to one
env a face n 0
run fall
env a face n 1
# fire without a fire list falls back to the critical frame
env v fire u
run fall
env v fire a 2 s "f0" s "f1"
# mfire1 keeps the direction unturned
env itr o 2 effect n 20 fall n 0
run fall
# mfire1 plus a held heavy weapon drops it first
env v holding o 1 base_type n 2
run fall
env v holding o 1 base_type n 3
env itr o 2 effect n 20 fall n 0
run fall
# fire explosion is position based, so the direction is always -1
env itr o 2 effect n 22 fall n 0
run fall
env a face n -1
run fall
env a face n 1
# fire explosion without a fire list falls back
env v fire u
run fall
env v fire a 2 s "f0" s "f1"
# unhandled effects go through the critical frame path
env itr o 2 effect n 3 fall n 0
run fall
env itr o 2 effect n 10000 fall n 0
run fall
# explosion is position based as well
env itr o 2 effect n 23 fall n 0
env velx 4
run fall
env velx 0
# cubes drive the spark point
env acube o 6 left n -3 right n -1 bottom n -7 top n -5 near n -9 far n -8
env bcube o 6 left n 100 right n 101 bottom n 102 top n 103 near n 104 far n 105
env itr o 2 effect n 0 fall n 0
run fall
