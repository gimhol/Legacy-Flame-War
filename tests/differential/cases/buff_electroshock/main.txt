# buff_electroshock
# `env id|kind|victim` take a string value; `env vframe|vpos` take an object.
# `env vtype|vstate` take a value literal; `env vwait|duration` take a bare number.
# `env victim` re-selects an existing victim (it is moved to the end of the list).
# run make|init|mount|unmount|tick <d>
# Buff_Electroshock: KIND="Electroshock", effect_oid="fx", effect_frame_id inherited ("0")
# StateEnum: Injured=11 Falling=12 Lying=14 ; EntityEnum: Fighter=8 Weapon=16
# init() sets the tick interval to 3; mount() halves the duration unless the victim state
# is Injured or Falling (loose ==); on_tick() bumps `wait` for fighters whose state is
# neither Falling nor Lying (strict ===).
env id s "B1"
env kind s "Electroshock"
env victim s "V1"
env vtype n 8
env vstate n 0
env vwait 0
env vpos o 3 x n 10 y n 20 z n 30
env vframe o 3 centery n 4 height n 10 pic_h n 0
run make
# ---- init() sets the tick interval, mount() halves the duration ----
env duration 8
run init
run mount
# ---- on_tick bumps `wait` ----
run tick 3
run tick 3
# ---- Falling and Lying states are skipped (strict) ----
env vstate n 12
run tick 3
env vstate n 14
run tick 3
# ---- a string "12" is not `===` Falling ----
env vstate s "12"
run tick 3
# ---- a string "14" is not `===` Lying ----
env vstate s "14"
run tick 3
# ---- a non fighter victim is skipped ----
env vstate n 0
env vtype n 16
run tick 3
env vtype n 8
# ---- the effect entity is centred and uses the base frame id ----
run tick 3
# ---- unmount ----
run unmount
# ---- mount leaves an Injured victim alone (numeric and string states) ----
env vstate n 11
env duration 8
run make
run mount
run unmount
env vstate s "11"
env duration 8
run make
run mount
run unmount
# ---- mount leaves a Falling victim alone (numeric and string states) ----
env vstate n 12
env duration 8
run make
run mount
run unmount
env vstate s "12"
env duration 8
run make
run mount
run unmount
# ---- a halved duration keeps its fraction ----
env vstate n 0
env duration 7
run make
run mount
run unmount
# ---- two victims are both processed ----
env victim s "V2"
env vtype n 8
env vstate n 0
env duration 8
run make
run mount
run unmount
