# character_state_burning
# `env victim` takes a string value; `env catcher` takes a bare number (0/1).
# `env indexes|wdata|facing|bounced|velx|vely|velz|landingvel|onlanding` take a value literal
#   (`velx` can be `u`).
# run make|enter|update|leave|landing
# StateEnum.Burning=18 ; world.dataset: cha_bc_tst_spd_y=5 cha_bc_tst_spd_x=3 cha_bc_spd=-3
# indexes: bouncing={"-1":["B0","B1"]} lying={"-1":"L-1","1":"L1"}
env victim s "V1"
env indexes o 2 bouncing o 1 -1 a 2 s "B0" s "B1" lying o 2 -1 s "L-1" 1 s "L1"
env wdata o 3 cha_bc_tst_spd_y n 5 cha_bc_tst_spd_x n 3 cha_bc_spd n -3
env facing n 1
env bounced n 0
env velx n 0
env vely n 0
env velz n 0
env catcher 0
env onlanding u
env landingvel o 2 x n 0 y n 0
run make
# ---- enter without a catcher ----
run enter
# ---- enter with a catcher ----
env catcher 1
env bounced n 1
run enter
# ---- update: a zero x velocity keeps the facing ----
env catcher 0
env velx n 0
run update
# ---- update: a positive x velocity turns the facing back ----
env velx n 5
run update
# ---- update: a negative x velocity turns the facing forward ----
env velx n -5
run update
# ---- update: a missing x velocity keeps the facing ----
env velx u
run update
# ---- leave resets the bounced flag ----
env bounced n 1
run leave
# ---- landing: the frame's own on_landing wins ----
env onlanding o 1 id s "20"
env bounced n 0
run landing
env onlanding u
# ---- landing: a slow enough fall bounces ----
env landingvel o 2 x n 0 y n 5
env bounced n 0
run landing
# ---- landing: a fast fall keeps lying ----
env landingvel o 2 x n 0 y n 6
env bounced n 0
run landing
# ---- landing: a large sideways speed bounces ----
env landingvel o 2 x n 4 y n 6
env bounced n 0
run landing
# ---- landing: a large negative sideways speed also bounces ----
env landingvel o 2 x n -4 y n 6
env bounced n 0
run landing
# ---- landing: an already bounced entity lies down ----
env bounced n 1
run landing
# ---- landing: a missing bounced value is falsy ----
env bounced u
env landingvel o 2 x n 0 y n 0
run landing
# ---- landing: a missing dataset poisons the test ----
env bounced n 0
env wdata o 0
run landing
