# burning_drink
# `env cls s "burning"|"drink"`; `env hp|hpr <bare number>`;
# `env catcher|onground|holding <truthy>`; the rest take value literals.
# run make|default|enter|update|leave|landing
# StateEnum: Drink=17 Burning=18; EntityEnum: Fighter=8 Weapon=16 Ball=32
# Ball_Hitting=3001
# `indexes.bouncing` is `{"-1":[_, id]}` (id at [1]); `lying` is `{"-1": id}`.
# `indexes.on_ground` must stay a string when a weapon landing is exercised:
# C++ renders `to_string(undefined)` as the string "undefined" while TS logs `u`.

# ============ State_Burning ============
env cls s "burning"
env state n 18
env data o 1 type n 8
env wdata o 3 cha_bc_tst_spd_y n 5 cha_bc_tst_spd_x n 3 cha_bc_spd n 9
env indexes o 2 bouncing o 1 "-1" a 2 u s "B5" lying o 1 "-1" s "L7"
env vstate n 18
env facing n 1
env vx n 0
env vy n 0
env vz n 0
run make
run default
# ---- fighter enter resets bounced and drops the catcher ----
env catcher b 1
env bounced b 1
run enter
# ---- fighter update flips facing against the velocity ----
env vx n 5
run update
env vx n -5
run update
# ---- a zero velocity does not flip the facing ----
env vx n 0
env facing n 1
run update
# ---- fighter leave clears bounced ----
env bounced b 1
run leave
# ---- a non-default state is kept by the constructor ----
env state n 7
run make
run default
# ---- landing: the frame wins ----
env state n 18
run make
env onlanding o 1 id s "55"
env vel o 3 x n 0 y n 0 z n 0
run landing
env onlanding u
# ---- landing: the bounce branch ----
env bounced u
env vel o 3 x n 0 y n -6 z n 0
run landing
# ---- landing: a truthy bounced blocks the bounce branch ----
env bounced b 1
run landing
# ---- landing: the speed-x branch skips the first dataset read ----
env bounced u
env vel o 3 x n 4 y n 100 z n 0
run landing
# ---- landing: the y threshold is inclusive ----
env bounced u
env vel o 3 x n 0 y n 5 z n 0
run landing
env vel o 3 x n 0 y n 6 z n 0
run landing
# ---- weapon data routes to the weapon proxy ----
env data o 1 type n 16
env indexes o 1 on_ground s "W9"
env onlanding u
env bounced u
run make
run update
run landing
# ---- ball data routes to the ball proxy ----
env data o 1 type n 32
env vstate n 3001
run make
run enter
# ---- other data routes to the plain base state ----
env data o 1 type n 4
env vstate n 18
run make
run update
run enter

# ============ CharacterState_Drink ============
env cls s "drink"
env state n 17
env data o 1 type n 8
run make
env state n 3
run make
run default
# ---- no holding: only the ground decay ----
env holding b 0
env drink u
env hp 50
env hpr 10
env hpmax n 60
env mp n 0
env mpmax n 100
run update
# ---- holding without a drink returns after the decay ----
env holding b 1
run update
# ---- every part empty: the drink is dropped ----
env drink o 3 hp_h_ticks n 5 hp_h_value n 0 hp_h_total n 10
env mtrange n 7
env hp 50
env hpr 30
env hpmax n 60
env mpmax n 100
run update
# ---- the hp ticks gate the recovery ----
env drink o 3 hp_h_ticks n 2 hp_h_value n 2 hp_h_total n 10
env mtrange u
env hp 50
env hpr 10
env hpmax n 60
run make
run update
run update
run update
run update
# ---- hp_r and mp recover with their own values ----
env drink o 6 hp_r_ticks n 1 hp_r_value n 3 hp_r_total n 10 mp_h_ticks n 1 mp_h_value n 4 mp_h_total n 10
env hp 50
env hpr 10
env hpmax n 55
env mp n 0
env mpmax n 6
run make
run update
run update
# ---- the hp_r ticks gate the recovery ----
env drink o 3 hp_r_ticks n 2 hp_r_value n 3 hp_r_total n 100
env hpr 10
env hpmax n 60
env mp n 0
env mpmax n 100
run make
run update
run update
# ---- the mp ticks gate the recovery ----
env drink o 3 mp_h_ticks n 2 mp_h_value n 4 mp_h_total n 100
run make
run update
run update
# ---- the hp recovery is clamped by hp_max ----
env drink o 3 hp_h_ticks n 1 hp_h_value n 10 hp_h_total n 100
env hp 50
env hpmax n 52
run make
run update
# ---- a missing hp_max poisons the recovery with NaN ----
env hpmax u
run update
env hpmax n 52
# ---- the hp_r recovery is clamped by hp_max too ----
env drink o 3 hp_r_ticks n 1 hp_r_value n 10 hp_r_total n 100
env hpr 10
env hp 5
env hpmax n 12
run make
run update
# ---- the mp recovery is clamped by mp_max ----
env drink o 3 mp_h_ticks n 1 mp_h_value n 4 mp_h_total n 100
env mp n 4
env mpmax n 6
run make
run update
run update
# ---- one empty part keeps the drink ----
env holding b 1
env drink o 3 hp_r_ticks n 1 hp_r_value n 3 hp_r_total n 100
env hp 50
env hpr 10
env hpmax n 60
env mtrange n 5
run make
run update
# ---- a zero total counts as empty ----
env drink o 3 hp_h_ticks n 1 hp_h_value n 5 hp_h_total n 0
env hp 50
run make
run update
