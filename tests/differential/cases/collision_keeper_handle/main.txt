# collision_keeper_handle
# `env a|v type|state` take a bare number; `env itr|bdy|handlers|data` take a value literal.
# This harness relies on the default actor ids (A / V).
# ItrKind: Normal=0 Pick=2 PickSecretly=7 MagicFlute=10 MagicFlute2=11 Block=14 Whirlwind=15
# `kind` must be present on itr and bdy; `action.type` selects the dispatched handler.
env itr o 1 kind n 0
env bdy o 1 kind n 0
env handlers a 1 s "handle_itr_normal_bdy_normal"
env data o 1 hit_sounds s "snd"
env a type 8
env v type 8
# ---- plain pass: no debug line, one handler call, no actions, hit sound ----
env dev 0
run hunt
# ---- the dev branch prints the collision description and the handler names ----
env dev 1
run hunt
env dev 0
# ---- handler order, including the two ball-is-hit names ----
env handlers a 3 s "handle_itr_catch" s "handle_ball_is_hit_a" s "handle_ball_is_hit_b"
run hunt
env handlers a 0
run hunt
env handlers a 1 s "handle_itr_catch"
# ---- pretested actions run their tester while the test list is built ----
env itr o 2 kind n 0 actions a 2 o 3 type s "A_NEXT_FRAME" pretest b 1 tester o 1 r b 1 o 3 type s "V_NEXT_FRAME" pretest b 0 tester o 1 r b 1
run hunt
# ---- a failing tester skips the action in both shapes ----
env itr o 2 kind n 0 actions a 2 o 3 type s "A_NEXT_FRAME" pretest b 1 tester o 1 r b 0 o 3 type s "V_NEXT_FRAME" pretest b 0 tester o 1 r b 0
run hunt
# ---- pretest without a tester always runs ----
env itr o 2 kind n 0 actions a 1 o 2 type s "A_NEXT_FRAME" pretest b 1
run hunt
# ---- no pretest and no tester always runs ----
env itr o 2 kind n 0 actions a 1 o 1 type s "A_NEXT_FRAME"
run hunt
# ---- bdy actions follow the same rules ----
env itr o 1 kind n 0
env bdy o 2 kind n 0 actions a 1 o 3 type s "V_NEXT_FRAME" pretest b 1 tester o 1 r b 1
run hunt
env bdy o 2 kind n 0 actions a 1 o 3 type s "V_NEXT_FRAME" pretest b 1 tester o 1 r b 0
run hunt
# ---- a missing actions value is ignored ----
env bdy o 1 kind n 0
env itr o 2 kind n 0 actions z
run hunt
# ---- six kinds keep the victim hit sound silent ----
env itr o 1 kind n 14
run hunt
env itr o 1 kind n 15
run hunt
env itr o 1 kind n 10
run hunt
env itr o 1 kind n 11
run hunt
env itr o 1 kind n 2
run hunt
env itr o 1 kind n 7
run hunt
env itr o 1 kind n 0
run hunt
env itr o 1 kind s "0"
run hunt
env itr o 1 kind s "14"
run hunt
# ---- a missing hit sound is undefined ----
env data o 0
run hunt
env data o 1 other s "x"
run hunt
# ---- dev line with distinct types and an unknown itr kind ----
env data o 1 hit_sounds s "snd"
env dev 1
env a type 8
env v type 32
env itr o 1 kind n 99
run hunt
env a type 8
env v type 8
env itr o 1 kind n 0
# ---- dev line with distinct itr and bdy enum names ----
env dev 1
env bdy o 1 kind n 2000
env itr o 1 kind n 16
run hunt
env dev 0
env bdy o 1 kind n 0
env itr o 1 kind n 0
# ---- both action lists present: the two test lists must not be swapped ----
env itr o 2 kind n 0 actions a 1 o 3 type s "A_NEXT_FRAME" pretest b 1 tester o 1 r b 1
env bdy o 2 kind n 0 actions a 1 o 3 type s "V_NEXT_FRAME" pretest b 1 tester o 1 r b 0
run hunt
env dev 0
env bdy o 1 kind n 0
env itr o 1 kind n 0
run hunt
