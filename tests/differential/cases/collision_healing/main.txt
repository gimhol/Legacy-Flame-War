# collision_healing
# `env a|v` lines carry exactly one field; trailing tokens are a hard error.
# Buff_Healing.KIND="Healing"; buff id = kind + "_" + victim.id (here "Healing_V")
# duration = ceil(injury / max(1, hp_healing_value)) * max(1, hp_healing_ticks)
# NOTE: `dataset` is only exercised with objects; a null `dataset` is out of contract
# (the real Entity.dataset() is an ordinary property read, it never throws).
env a dataset o 0
env v dataset o 2 hp_healing_value n 10 hp_healing_ticks n 1
# ---- falsy injury short circuits before any entity lookup ----
env itr o 1 injury n 0
run heal
env itr o 1 injury s ""
run heal
env itr o 1 injury u
run heal
env itr o 0
run heal
env itr o 1 injuries n 5
run heal
# ---- happy paths ----
env itr o 1 injury n 15
run heal
env itr o 1 injury n 20
run heal
env itr o 1 injury n 21
run heal
# ---- ticks multiply the ceiling ----
env v dataset o 2 hp_healing_value n 10 hp_healing_ticks n 3
env itr o 1 injury n 21
run heal
env itr o 1 injury n 1
run heal
# ---- hp_healing_value clamps up to 1 ----
env v dataset o 2 hp_healing_value n 0 hp_healing_ticks n 1
env itr o 1 injury n 5
run heal
env v dataset o 2 hp_healing_value n 1 hp_healing_ticks n 1
run heal
env v dataset o 2 hp_healing_value n 0.5 hp_healing_ticks n 2
env itr o 1 injury n 5
run heal
# ---- hp_healing_ticks clamps up to 1 ----
env v dataset o 2 hp_healing_value n 10 hp_healing_ticks n 0
env itr o 1 injury n 25
run heal
env v dataset o 2 hp_healing_value n 10 hp_healing_ticks n 1
run heal
# ---- missing dataset keys become NaN ----
env v dataset o 0
env itr o 1 injury n 7
run heal
env v dataset o 1 hp_healing_value n 10
run heal
env v dataset o 1 hp_healing_ticks n 2
run heal
# ---- non object dataset is out of contract; empty object is the closest input ----
env v dataset o 0
env itr o 1 injury n 7
run heal
# ---- injury coercion ----
env v dataset o 2 hp_healing_value n 10 hp_healing_ticks n 2
env itr o 1 injury s "6"
run heal
env itr o 1 injury s "abc"
run heal
env itr o 1 injury n -9
run heal
env itr o 1 injury n 3.5
run heal
