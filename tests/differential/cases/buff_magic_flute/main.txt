# buff_magic_flute
# `env id|kind|cls|attacker|victim` take a string value.
# `env vtype|vstate|vhp|vhp_r|vfallinjury|vtough|vindexes` take a value literal.
# `env ateam|vteam` take a string value: `Entity.team` is a `string` (TS
# `get team(): string`), a number would not survive the identity round-trip.
# `env vvy` takes a bare number. `env victim` re-selects an existing victim.
# `env attacker s ""` clears the attacker.
# run make|init|tick <d>
# EntityEnum: Fighter=8 Weapon=16 Ball=4 ; StateEnum: Falling=12 Weapon_InTheSky=1000
#   HeavyWeapon_InTheSky=2000 ; SpeedMode.AccTo=4
# Buff_MagicFlute: injury=2, injury_r=0.5, calc_v(y, 3, AccTo, 3, 1)
# Buff_MagicFlute2: injury=1, injury_r=0.5, calc_v(y, 1.5, AccTo, 1.5, 1)
# indexes.falling = {"-1": ["30","31"], "1": ["35"]} ; indexes.in_the_skys = ["40","41"]
env cls s "mf1"
env id s "B1"
env kind s "MagicFlute"
env attacker s "A1"
env ateam s "1"
env victim s "V1"
env vtype n 8
env vindexes o 2 falling o 2 -1 a 2 s "30" s "31" 1 a 1 s "35" in_the_skys a 2 s "40" s "41"
env vstate n 0
env vhp n 10
env vhp_r n 2
env vfallinjury n 0
env vtough n 0
env vteam s "2"
env vvy 0
run make
run init
# ---- vy accelerates toward the target by `acc` ----
run tick 3
# ---- vy beyond the target is kept ----
env vvy 5
run tick 3
# ---- vy equal to the target is kept ----
env vvy 3
run tick 3
# ---- a negative vy still moves by `acc` ----
env vvy -5
run tick 3
# ---- Fighter in Falling keeps its frame ----
env vstate n 12
env vvy 0
run tick 3
# ---- a string "12" is not `===` Falling ----
env vstate s "12"
run tick 3
# ---- a missing attacker keeps the damage summary silent ----
env attacker s ""
env vstate n 0
env vhp n 10
env vhp_r n 2
env vvy 0
run make
run tick 3
# ---- Weapon already in the sky is left alone ----
env attacker s "A1"
env vtype n 16
env vstate n 1000
env vhp n 10
env vhp_r n 2
env vvy 0
run make
run tick 3
env vstate n 2000
run tick 3
# ---- a Weapon on the ground adopts the attacker team and the sky frame ----
env vstate n 0
run tick 3
# ---- without an attacker only the frame changes ----
env attacker s ""
run tick 3
# ---- every other entity type is ignored ----
env attacker s "A1"
env vtype n 4
env vstate n 0
env vhp n 10
env vhp_r n 2
run tick 3
# ---- a string "8" is not `===` Fighter ----
env vtype s "8"
env vstate n 0
run tick 3
# ---- a string "16" is not `===` Weapon ----
env vtype s "16"
env vstate n 0
run tick 3
# ---- the second class halves nothing and hurts by one ----
env cls s "mf2"
env kind s "MagicFlute2"
env vtype n 8
env vstate n 0
env vhp n 10
env vhp_r n 2
env vvy 0
env vfallinjury n 0
env vtough n 5
run make
run init
run tick 3
# ---- every victim is visited in order ----
env victim s "V2"
env vtype n 8
env vstate n 12
env vhp n 10
env vhp_r n 2
env vvy -2
run make
run tick 3
