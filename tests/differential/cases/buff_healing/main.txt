# buff_healing
# `env id|kind|cls|victim` take a string value; `env vdata|vhp|vmp|mark` take an object.
# `env ticks|duration` take a bare number and require a buff (run make first).
# run make|mount|unmount|tick <d>|duration <amount>
# Buff_Healing.KIND="Healing" Buff_MpHealing.KIND="MpHealing"
# duration_of = ceil(amount / max(1,<kind>_healing_value)) * max(1,<kind>_healing_ticks)
# The ticker accumulates: on_tick fires only once the accumulated value reaches `ticks`,
# so `run tick <ticks>` fires on every call while `run tick 1` fires once per `ticks` calls.
env cls s "healing"
env kind s "Healing"
env id s "B1"
env victim s "V1"
env vdata o 4 hp_healing_value n 10 hp_healing_ticks n 2 mp_healing_value n 5 mp_healing_ticks n 3
env vhp o 2 hp n 5 hp_r n 50
env vmp o 2 mp n 2 mp_max n 9
# ---- duration_of: ceiling, exact division, zero and negative amounts ----
run duration 25
run duration 20
run duration 1
run duration 0
run duration -3
# ---- the value clamp lifts 0 up to 1 ----
env vdata o 4 hp_healing_value n 0 hp_healing_ticks n 2 mp_healing_value n 0 mp_healing_ticks n 1
run duration 5
# ---- the ticks clamp lifts 0 up to 1 ----
env vdata o 4 hp_healing_value n 10 hp_healing_ticks n 0 mp_healing_value n 5 mp_healing_ticks n 0
run duration 25
env vdata o 4 hp_healing_value n 10 hp_healing_ticks n 2 mp_healing_value n 5 mp_healing_ticks n 3
# ---- make + mount marks the victim and copies the tick interval ----
run make
run mount
run mount
# ---- on_tick: three calls of 1 fire exactly once ----
run tick 1
run tick 1
run tick 1
# ---- hp near the cap: the heal must be clamped to hp_r ----
env vhp o 2 hp n 45 hp_r n 50
run tick 2
run tick 2
# ---- hp already at the cap only refreshes the lifetime ----
env vhp o 2 hp n 50 hp_r n 50
run tick 2
# ---- unmount removes the mark ----
run unmount
# ---- a foreign mark value must not be removed ----
env vhp o 2 hp n 5 hp_r n 50
env victim s "V2"
env vdata o 4 hp_healing_value n 4 hp_healing_ticks n 1 mp_healing_value n 5 mp_healing_ticks n 3
env vhp o 2 hp n 0 hp_r n 10
run make
run mount
env mark o 2 key s "Healing" value s "FOREIGN"
run unmount
# ---- mp healing keeps the same shape with another dataset ----
env cls s "mp_healing"
env kind s "MpHealing"
env id s "B2"
run duration 26
run duration 2
env vmp o 2 mp n 2 mp_max n 9
run make
run mount
run tick 3
run tick 3
run tick 3
env vmp o 2 mp n 9 mp_max n 9
run tick 3
env mark o 2 key s "MpHealing" value s "FOREIGN"
run unmount
