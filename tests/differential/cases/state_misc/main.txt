# state_misc
# `env cls s "weapon_broken"|"to_catching"|"to_louisex"|"to_8xxx"|"ball"`.
# `env victim` takes a string value; `env state|vstate|vtype|finddata|findfighter|transformtype`
#   take a value literal.
# run make|landing|update|enter|leave
# StateEnum: Ball_Hitting=3001 Ball_Hit=3002 Ball_Rebounding=3003 Ball_Disappear=3004
#   Weapon_Brokens=9999 TurnIntoLouisEX=9995 TransformToCatching_End=501
# EntityEnum.Fighter=8
env victim s "V1"
env vstate n 0
env vtype n 4
env finddata n 1
env findfighter n 1
env transformtype n 8
# ---- weapon broken: landing enters the gone frame ----
env cls s "weapon_broken"
env state n 9999
run make
run landing
# ---- transform to catching: update transforms and enters the auto frame ----
env cls s "to_catching"
env state n 501
run make
run update
# ---- transform to Louis EX: enter transforms when the fighter data exists ----
env cls s "to_louisex"
env state n 9995
env vtype n 4
run make
run enter
# ---- transform to Louis EX: a missing fighter data does nothing ----
env findfighter u
run enter
# ---- transform to 8XXX: leave transforms and reports a new fighter ----
env cls s "to_8xxx"
env state n 8008
env vtype n 4
env transformtype n 8
env finddata n 1
run make
run leave
# ---- transform to 8XXX: the same type is not reported ----
env vtype n 8
run leave
# ---- transform to 8XXX: another type is not reported ----
env vtype n 4
env transformtype n 16
run leave
# ---- transform to 8XXX: a string "8" is not === Fighter ----
env vtype n 4
env transformtype s "8"
env finddata n 1
run leave
# ---- transform to 8XXX: a missing new type is not reported ----
env transformtype n 8
env finddata u
run leave
# ---- transform to 8XXX: a string state does nothing ----
env state s "8008"
run make
run leave
# ---- ball base: enter zeroes the flags only for the four ball states ----
env cls s "ball"
env state n 0
env vstate n 3001
run make
run enter
env vstate n 3002
run enter
env vstate n 3003
run enter
env vstate n 3004
run enter
env vstate n 3000
run enter
env vstate s "3001"
run enter
