# weapon_state_misc
# `env cls s "onground"|"onhand"|"throwing"|"inthesky"`; `env hp|hpr <bare number>`;
# `env bearer|dropping <truthy>`; the rest take value literals.
# run make|enter|update|landing|preupdate|gravity
# StateEnum: Weapon_InTheSky=1000 Weapon_OnHand=1001 Weapon_Throwing=1002 Weapon_OnGround=1004
# WT.Heavy=2 FrameBehavior.Boomerang=3
# Landing with an undefined nf is avoided: C++ renders the id as the string
# "undefined", TS renders the raw undefined.
# `base` must stay an object whenever `update`/`landing` reach
# `hit_ground_rebouncing`: TS reads `base.bounce_x` without `?.`.

# ============ WeaponState_OnGround ============
env cls s "onground"
env state n 1004
env newteam s "team_7"
env team s "team_1"
run make
# ---- enter replaces the team ----
run enter
# ---- a missing new_team is written as undefined ----
env newteam u
run enter
# ---- update only decays ----
run update

# ============ WeaponState_OnHand ============
env cls s "onhand"
env state n 1001
run make
# ---- the bearer takes the larger motionless ----
env motionless n 5
env bearer b 1
env bmotion n 2
run preupdate
# ---- a falsy motionless writes nothing ----
env motionless n 0
env bmotion n -3
run preupdate
# ---- without a bearer nothing is written ----
env motionless n 5
env bearer b 0
run preupdate
# ---- NaN propagates through max ----
env bearer b 1
env motionless s "abc"
env bmotion n 4
run preupdate
# ---- numeric strings are coerced ----
env motionless s "4"
env bmotion s "6"
run preupdate
# ---- the bearer keeps the larger value ----
env motionless n 9
env bmotion n 3
run preupdate

# ============ WeaponState_Throwing ============
env cls s "throwing"
env state n 1002
env behavior n 3
env gval n 10
run make
# ---- gravity: the boomerang quarter ----
run gravity
# ---- gravity: rounding to the 1/1000 grid ----
env gval n 1.2345
run gravity
# ---- gravity: no quarter without the behavior ----
env behavior n 2
run gravity
# ---- gravity: the behavior comparison is loose ----
env behavior s "3"
run gravity
# ---- gravity: a missing dataset stays NaN ----
env behavior n 3
env gval u
run gravity
# ---- gravity: a numeric string is divided ----
env gval s "8"
run gravity
# ---- enter: drop hurted is reset, velocity is scaled ----
env vx n 7
env dh b 1
run enter
# ---- enter: no scaling without the behavior ----
env behavior n 2
env dh b 1
run enter
# ---- enter: a missing behavior still leaves ground and resets ----
env behavior u
env dh u
run enter
# ---- update only decays ----
run update
# ---- landing: the frame wins ----
env onlanding o 1 id s "55"
env vel o 3 x n 0 y n 0 z n 0
run landing
env onlanding u
# ---- landing: throw_on_ground wins over just_on_ground ----
env base o 0
env indexes b 1
env throwg s "TG"
env justg s "JG"
run landing
# ---- landing: a falsy throw_on_ground falls through ----
env throwg u
run landing
env throwg n 0
run landing
# ---- landing: just_on_ground alone ----
env justg u
env throwg s "TG"
run landing
# ---- landing: the bounce branch is inherited ----
env throwg s "TG"
env justg u
env vstate n 1002
env align o 1 id s "88"
env fid s "F1"
env isky a 1 s "SKY"
env ithrow a 1 s "THR"
env vel o 3 x n 0 y n -10 z n 0
run landing

# ============ WeaponState_InTheSky ============
env cls s "inthesky"
env state n 1000
run make
env base o 0
# ---- enter resets drop_hurted ----
env dh b 1
run enter
# ---- update: nothing is fast enough yet ----
env vx u
env vy u
env vz u
run update
# ---- update: heavy weapons never turn into throwing ----
env wt n 2
env vx n 1000
run update

env indexes b 0

env vx n 5
env vy n 0
env vz n 0
run update

env isky a 1 s "SKY"

run update

env dropping b 1
run update
# ---- update: without an align frame nothing is entered ----

env dropping b 1
run update
# ---- update: base thresholds win over the table ----

env base o 1 fast_vx n 10
env vx n 5
run update
env vx n 11
run update

env vz n 0
