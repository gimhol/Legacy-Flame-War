# character_state_caught_rowing
# `env cls s "caught"|"rowing"`; `env victim` takes a string value.
# `env has_holding` takes a bare number (0/1).
# `env state|prevstate|fall|fallmax|vteam|velx|vely|holding|onlanding|dataset|indexes` take a value
#   literal (`velx`/`vely`/`fall`/`fallmax` accept `u`).
# run make|enter|update|landing
# StateEnum: Caught=10 Falling=12 Rowing=6 ; WeaponEnum.Heavy=2 ; SpeedMode.Default=0
# `prevstate` is the `state` field of the frame passed to `enter`.
env victim s "V1"
env cls s "caught"
env state n 10
env prevstate n 12
env fall n 1
env fallmax n 9
env vteam n 3
env velx n 7
env vely n 5
env has_holding 1
env holding n 2
env onlanding u
env dataset o 4 rowing_distance n 10 bfall_x_f n 0.5 rowing_height n 4 bfall_h_f n -3
env indexes o 2 landing_1 s "40" default s "50"
run make
# ---- caught enter: the fall value is reset, the velocity zeroed, the heavy holding adopted ----
run enter
# ---- caught enter: a non heavy holding is dropped but keeps its team ----
env fall n 1
env holding n 1
run enter
# ---- caught enter: no holding at all ----
env fall n 1
env has_holding 0
env holding n 2
run enter
# ---- caught enter: a string "2" holding is not === Heavy ----
env fall n 1
env has_holding 1
env holding s "2"
run enter
# ---- caught update: the velocity is zeroed and nothing else runs ----
env velx n 7
env vely n 5
run update
# ---- rowing enter: only a falling previous frame reacts ----
env cls s "rowing"
env state n 6
env prevstate n 12
env velx n 7
env vely n 5
run make
run enter
# ---- rowing enter: a positive velocity keeps the rowing distance positive ----
env velx n 0
env vely n 5
run enter
# ---- rowing enter: a negative velocity flips the rowing distance ----
env velx n -3
env vely n -5
run enter
# ---- rowing enter: a missing x velocity flips the rowing distance ----
env velx u
env vely n -5
run enter
# ---- rowing enter: a very negative y velocity is kept ----
env velx n 2
env vely n -20
run enter
# ---- rowing enter: another previous state does nothing ----
env prevstate n 0
run enter
# ---- rowing enter: a string "12" previous state is not === Falling ----
env prevstate s "12"
run enter
# ---- rowing landing: the frame's own on_landing wins ----
env prevstate n 12
env onlanding o 1 id s "20"
run landing
env onlanding u
# ---- rowing landing: otherwise indexes.landing_1 is used ----
run landing
# ---- rowing update: only the base class runs ----
run update
