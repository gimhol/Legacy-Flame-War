# make_itr_prefabs：块提取 + 逐行 entry 匹配 + 冒号键值 + cook_itr
new p1
set p1 text s "<weapon_strength_list>\u000Aentry:1 Foo\u000Ahp: 100 x: 5\u000Aentry:2 Bar\u000Amass: 3\u000A<weapon_strength_list_end>"
ip p1
# 块外内容不算；块内容前后空白会被 trim
new p2
set p2 text s "junk\u000A<weapon_strength_list>\u000A\u0009entry:1 Foo\u000A<weapon_strength_list_end>\u000Atail"
ip p2
# 无块 / 块内只有空白 / 非字符串
new p3
set p3 text s "no markers here"
ip p3
new p4
set p4 text s "<weapon_strength_list>\u000A\u000A<weapon_strength_list_end>"
ip p4
new p5
set p5 text n 7
ip p5
# entry 行内没有名字：\d+ 会回溯，名字落到下一行（\s* 跨换行）
new p6
set p6 text s "<weapon_strength_list>\u000Aentry: 123\u000Axyz: 1\u000A<weapon_strength_list_end>"
ip p6
# 数字与名字在同一段连续非空白里：id 取最长数字段，名字是剩下的
new p7
set p7 text s "<weapon_strength_list>\u000Aentry:12 34\u000A<weapon_strength_list_end>"
ip p7
# 空行被 \s* 跳过；remain 是紧随其后的那一行
new p8
set p8 text s "<weapon_strength_list>\u000Aentry:1 Foo\u000A\u000A\u000Ahp: 7\u000A<weapon_strength_list_end>"
ip p8
# 冒号键覆盖 entry.id：最终映射的键取覆盖后的 id
new p9
set p9 text s "<weapon_strength_list>\u000Aentry:1 Foo\u000Aid: 42 kind: 2\u000A<weapon_strength_list_end>"
ip p9
# 重复 id：后者覆盖，键位置保持首次出现
new p10
set p10 text s "<weapon_strength_list>\u000Aentry:1 A\u000A\u000Aentry:1 B\u000A<weapon_strength_list_end>"
ip p10
# 冒号值里含冒号（全局迭代）与空值
new p11
set p11 text s "<weapon_strength_list>\u000Aentry:1 A\u000Aurl: b:c empty:\u000A<weapon_strength_list_end>"
ip p11
# 只有 entry 行、没有后续行
new p12
set p12 text s "<weapon_strength_list>\u000Aentry:1 Foo\u000A<weapon_strength_list_end>"
ip p12
# 数字段缺失的 entry 行不构成匹配（\d+ 要求至少一位），扫描继续往后
new p13
set p13 text s "<weapon_strength_list>\u000Aentry:abc hp: 1\u000Aentry:2 B\u000A<weapon_strength_list_end>"
ip p13
# 块非空但没有任何 entry 行（list 为空 ⇒ undefined，不是空对象）
new p14
set p14 text s "<weapon_strength_list>\u000Ajust some text\u000A<weapon_strength_list_end>"
ip p14
