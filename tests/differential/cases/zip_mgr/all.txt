# zip_mgr / 单个数据包：命中、未命中、多路径顺序、exact 去重差别
zip z1 "pack-a"
zfile z1 "a.dat" hit "a.dat"
zfile z1 "b.dat" hit "b.dat"
info i1
imd5 i1 s "md5-a"
add z1 i1
dump
find 1 p 1 "a.dat"
find 1 p 1 "c.dat"
find 1 p 2 "a.dat" "b.dat"
find 1 p 2 "b.dat" "a.dat"
find 1 p 2 "a.dat" "a.dat"
find 0 p 1 "a.dat"
find 0 p 2 "a.dat" "b.dat"
find 1 p 3 "c.dat" "b.dat" "a.dat"
