# character_state_base
# `env victim` takes a string value; `env onground` takes a bare number.
# `env state` is the state object's own state; `env vstate` is the entity's state.
# `env state|vstate|hp|facing|holding|onlanding|dataset|indexes|frames|velx|velz` take a value
#   literal (`holding`/`onlanding`/`velx`/`velz` accept `u`).
# run make|update|landing|up|auto|sudden|caught
# StateEnum: Standing=0 Walking=1 Running=2 Rowing=6 Falling=12 Weapon_InTheSky=1000
# WeaponEnum.Heavy=2
# indexes: landing_2="40" default="50" heavy_obj_walk="60" in_the_skys=["70","71"]
#   falling={"1":["80","81"],"-1":["90","91"]} ; frames: 40/50/60/70 are known
env victim s "V1"
env state n 0
env vstate u
env hp n 10
env facing n 1
env onground 1
env holding u
env onlanding u
env dataset o 2 cvx_d n 3 cvy_d n -4
env indexes o 5 landing_2 s "40" default s "50" heavy_obj_walk s "60" in_the_skys a 2 s "70" s "71" falling o 2 1 a 2 s "80" s "81" -1 a 2 s "90" s "91"
env frames o 4 40 s "F40" 50 s "F50" 60 s "F60" 70 s "F70"
env velx n 0
env velz n 0
run make
# ---- update decays the ground velocity ----
run update
# ---- landing without an on_landing frame uses indexes.landing_2 ----
run landing
# ---- the frame's own on_landing wins ----
env onlanding o 1 id s "20"
run landing
env onlanding u
# ---- auto frame: on the ground uses indexes.default ----
run auto
# ---- auto frame: airborne with hp uses indexes.in_the_skys[0] ----
env onground 0
run auto
# ---- auto frame: airborne with hp <= 0 yields nothing ----
env hp n 0
run auto
# ---- auto frame: a heavy holding wins over standing on the ground ----
env hp n 10
env onground 1
env holding n 2
run auto
# ---- auto frame: a holding that is not heavy is ignored ----
env holding n 1
run auto
# ---- auto frame: a string "2" is not `===` Heavy ----
env holding s "2"
run auto
# ---- auto frame: a missing default key yields nothing ----
env holding u
env indexes o 4 landing_2 s "40" default s "50" heavy_obj_walk s "60" in_the_skys a 2 s "70" s "71"
run auto
env indexes o 5 landing_2 s "40" default s "50" heavy_obj_walk s "60" in_the_skys a 2 s "70" s "71" falling o 2 1 a 2 s "80" s "81" -1 a 2 s "90" s "91"
# ---- sudden death: velocity is 2 * facing and the frame is falling["1"][1] ----
env facing n -1
env velx n 7
run sudden
# ---- caught end: velocity is -cvx_d * facing and the frame is falling["-1"][1] ----
run caught
# ---- caught end with an empty dataset ----
env dataset o 0
run caught
# ---- leave ground: running with a heavy holding drops it ----
env vstate n 2
env holding n 2
env dataset o 2 cvx_d n 3 cvy_d n -4
run up
# ---- leave ground: a holding of another type is kept ----
env holding n 1
run up
# ---- leave ground: a missing holding is kept ----
env holding u
run up
# ---- leave ground: a string state is not `===` Running ----
env holding n 2
env vstate s "2"
run up
# ---- leave ground: walking, standing and rowing also react ----
env holding u
env vstate n 1
run up
env vstate n 0
run up
env vstate n 6
run up
# ---- leave ground: every other state does nothing ----
env vstate n 12
run up
env vstate n 1000
run up
# ---- sudden death and caught end without a falling list yield nothing ----
env vstate n 0
env indexes o 4 landing_2 s "40" default s "50" heavy_obj_walk s "60" in_the_skys a 2 s "70" s "71"
run sudden
run caught
