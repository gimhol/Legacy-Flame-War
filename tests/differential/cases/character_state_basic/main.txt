# character_state_basic
# `env cls` selects standing|running|injured; `env usedefault 1` builds without a state.
# `env victim` takes a string value; `env onground|usedefault` take a bare number.
# `env state|vstate|hp|facing|vteam|ground_y|holding|onlanding|dataset|indexes|frames|pos|velx|velz`
#   take a value literal (`holding`/`velx`/`velz` accept `u`).
# run make|update|enter
# StateEnum: Standing=0 Running=2 Injured=11 ; WeaponEnum.Heavy=2
# The entity side supplies `get_sudden_death_frame` as {id:"SD"} and logs the call.
env victim s "V1"
env vstate n 0
env hp n 10
env facing n 1
env vteam n 3
env onground 1
env ground_y n 0
env holding u
env onlanding u
env indexes o 2 default s "50" in_the_skys a 2 s "70" s "71"
env pos o 3 x n 1 y n 0 z n 2
env velx u
env velz u
# ---- Standing: alive and on the ground does nothing ----
env cls s "standing"
run make
run update
# ---- Standing: above the ground enters the sky frame ----
env pos o 3 x n 1 y n 5 z n 2
env ground_y n 4
run update
# ---- Standing: exactly on the ground does nothing ----
env pos o 3 x n 1 y n 4 z n 2
run update
# ---- Standing: a dead entity enters the sudden death frame ----
env hp n 0
run update
# ---- Standing: death wins over the sky frame ----
env hp n 0
env pos o 3 x n 1 y n 5 z n 2
run update
# ---- Standing: a missing hp still counts as alive ----
env hp u
run update
# ---- Standing: the default state is used when none is given ----
env usedefault 1
run make
run update
env usedefault 0
# ---- Standing: an explicit state overrides the default ----
env hp n 10
env state n 2
run make
run update
# ---- Running: a zero z velocity leaves x alone ----
env cls s "running"
env state n 2
env hp n 10
env velx n 10
env velz n 0
run make
run update
# ---- Running: x above the z drag is reduced by it ----
env velx n 10
env velz n 8
run update
# ---- Running: x below the negative drag is raised by it ----
env velx n -10
env velz n 8
run update
# ---- Running: x inside the drag keeps its sign ----
env velx n 1
env velz n 8
run update
# ---- Running: a negative z drag uses its absolute value ----
env velx n 10
env velz n -8
run update
# ---- Running: a missing x stays missing ----
env velx u
env velz n 4
run update
# ---- Running: a missing z performs no drag at all ----
env velx n 10
env velz u
run update
# ---- Running: a dead entity enters the sudden death frame ----
env hp n 0
env velx n 10
env velz n 4
run update
# ---- Injured: a heavy holding is dropped and adopted ----
env cls s "injured"
env state n 11
env hp n 10
env holding n 2
run make
run enter
# ---- Injured: another holding type is untouched ----
env holding n 1
run enter
# ---- Injured: a string "2" is not `===` Heavy ----
env holding s "2"
run enter
# ---- Injured: a missing holding is untouched ----
env holding u
run enter
# ---- Injured: the default state is used when none is given ----
env usedefault 1
env holding n 2
run make
run enter
# ---- Injured: update only drives the base class ----
run update
