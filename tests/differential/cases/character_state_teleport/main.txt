# character_state_teleport
# `env cls s "nearest"|"farthest"`; `env victim` takes a string value;
# `env gy` takes a bare number; `env state|facing|seg|pos|ent` take a value literal.
# run make|enter
# StateEnum: TeleportToNearestEnemy=400 TeleportToFarthestAlly=401 ; EntityEnum.Fighter=8
# The distance is abs(o.x - m.x) + abs(o.z - o.z) -- the z term is always zero.
env victim s "M1"
env state n 400
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env seg n 7
env gy 3
env cls s "nearest"
run make
# ---- no candidates at all ----
run enter
# ---- a single enemy ----
env ent o 6 id s "E1" fighter b 1 ally b 0 hp n 10 x n 5 z n 2
run enter
# ---- every filter is exercised ----
env ent o 6 id s "E2" fighter b 0 ally b 0 hp n 10 x n 50 z n 0
env ent o 6 id s "E3" fighter b 1 ally b 1 hp n 10 x n 60 z n 0
env ent o 6 id s "E4" fighter b 1 ally b 0 hp n 0 x n 70 z n 0
run enter
# ---- the nearest enemy wins ----
env ent o 6 id s "E5" fighter b 1 ally b 0 hp n 10 x n 3 z n 0
run enter
# ---- the own entity is skipped even when it is the nearest ----
env pos o 3 x n 1 y n 0 z n 0
env ent o 6 id s "M1" fighter b 1 ally b 0 hp n 10 x n 1 z n 0
run enter
# ---- a backward facing offsets the other way ----
env pos o 3 x n 0 y n 0 z n 0
env facing n -1
env ent o 6 id s "E5" fighter b 1 ally b 0 hp n 10 x n 30 z n 0
run enter
# ---- the rounding of a fractional target ----
env facing n 1
env ent o 6 id s "E5" fighter b 1 ally b 0 hp n 10 x n 30.6 z n 4.4
run enter
# ---- the farthest ally picks the opposite end ----
env cls s "farthest"
env state n 401
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env ent o 6 id s "A1" fighter b 1 ally b 1 hp n 10 x n 5 z n 0
env ent o 6 id s "A2" fighter b 1 ally b 1 hp n 10 x n 40 z n 0
run make
run enter
# ---- an out of range ally is skipped ----
env ent o 6 id s "A2" fighter b 1 ally b 1 hp n 0 x n 40 z n 0
run enter
# ---- with no ally the own position is kept ----
env ent o 6 id s "A1" fighter b 1 ally b 0 hp n 10 x n 5 z n 0
run enter
# ---- an empty candidate list keeps the own position ----
env cls s "nearest"
env victim s "M2"
env facing n 1
env pos o 3 x n 9 y n 0 z n 8
run make
run enter
# ---- a dead enemy is filtered out even when it is the nearest ----
env victim s "M3"
env cls s "nearest"
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env ent o 6 id s "D1" fighter b 1 ally b 0 hp n 0 x n 1 z n 0
env ent o 6 id s "D2" fighter b 1 ally b 0 hp n 10 x n 30 z n 0
run make
run enter
# ---- a non fighter is skipped even when it is the nearest ----
env victim s "M9"
env cls s "nearest"
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env ent o 6 id s "N1" fighter b 0 ally b 0 hp n 10 x n 1 z n 0
env ent o 6 id s "N2" fighter b 1 ally b 0 hp n 10 x n 20 z n 0
run make
run enter
# ---- the z term of the distance is dead in the original ----
env victim s "M4"
env cls s "nearest"
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env ent o 6 id s "Z1" fighter b 1 ally b 0 hp n 10 x n 6 z n 100
env ent o 6 id s "Z2" fighter b 1 ally b 0 hp n 10 x n 7 z n 0
run make
run enter
# ---- equidistant enemies keep the first one ----
env victim s "M5"
env cls s "nearest"
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env ent o 6 id s "T1" fighter b 1 ally b 0 hp n 10 x n 5 z n 0
env ent o 6 id s "T2" fighter b 1 ally b 0 hp n 10 x n -5 z n 0
run make
run enter
# ---- equidistant allies keep the first one for the farthest search ----
env victim s "M7"
env cls s "farthest"
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env ent o 6 id s "V1" fighter b 1 ally b 1 hp n 10 x n 5 z n 0
env ent o 6 id s "V2" fighter b 1 ally b 1 hp n 10 x n -5 z n 0
run make
run enter
# ---- the last candidate can win ----
env victim s "M8"
env cls s "nearest"
env facing n 1
env pos o 3 x n 0 y n 0 z n 0
env ent o 6 id s "W1" fighter b 1 ally b 0 hp n 10 x n 40 z n 0
env ent o 6 id s "W2" fighter b 1 ally b 0 hp n 10 x n 2 z n 0
run make
run enter
# ---- the own x takes part in the distance ----
env victim s "MC"
env cls s "nearest"
env facing n 1
env ent o 6 id s "C1" fighter b 1 ally b 0 hp n 10 x n 6 z n 0
env ent o 6 id s "C2" fighter b 1 ally b 0 hp n 10 x n -7 z n 0
env pos o 3 x n 0 y n 0 z n 0
run make
run enter

run enter
