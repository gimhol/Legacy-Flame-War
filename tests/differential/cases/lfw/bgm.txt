# 4BL：BGMSwitcher

# ---- 0) bgms 列表 + 节点/组件（args 空 = offset 0）----
setbgms a 2 s "a.mp3" s "b.mp3"
bgms
nmk sw - o 6 id s "sw" pos a 3 n 0 n 0 n 0 size a 3 n 100 n 20 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 4 cls s "BGMSwitcher" id s "swc" args a 0 properties o 0
nrd sw text
nlife sw start
nlife sw resume
nrd sw text

# ---- 1) 左键 +1：Silent → Random（随机挑一个）→ a.mp3 → b.mp3 → 回到 Silent ----
nclick sw 0
nrd sw text
sndbgm
nclick sw 0
nrd sw text
sndbgm
nclick sw 0
nrd sw text
sndbgm
nclick sw 0
nrd sw text
sndbgm

# ---- 2) 右键 -1：Silent → b.mp3 → a.mp3 ----
nclick sw 2
nrd sw text
nclick sw 2
nrd sw text
sndbgm

# ---- 3) 中键：非 Silent → Silent；Silent → Random ----
nclick sw 1
nrd sw text
nclick sw 1
nrd sw text
sndbgm

# ---- 4) 不认识的按键：直接返回（但两端 stop 已经做过）----
nclick sw 3
nrd sw text

# ---- 5) offset 组件：resume 只继承 is_random（不注册不写文本）；点击不看按键 ----
nmk sw2 - o 6 id s "sw2" pos a 3 n 0 n 20 n 0 size a 3 n 100 n 20 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 4 cls s "BGMSwitcher" id s "sw2c" args a 1 n 1 properties o 0
nlife sw2 start
nlife sw2 resume
nrd sw2 text
nclick sw2 3
nrd sw2 text
nrd sw text
sndbgm

# ---- 6) 暂停退订：点击不再更新文本 ----
nlife sw pause
nclick sw 0
nrd sw text
sndbgm

# ---- 7) 空 bgms 下播 Random：静默（无状态、无回调）----
setbgms a 0
nlife sw resume
nrd sw text
nclick sw 1
nrd sw text
nclick sw 1
nrd sw text
bgms
sndbgm
