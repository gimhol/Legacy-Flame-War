# ---- StageTransitions：resume 订、pause 退（重复调用不叠加）----
nmk t1 - o 6 id s "t1" pos a 3 n 0 n 0 n 0 size a 3 n 10 n 10 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 3 cls s "StageTransitions" id s "t1c" args a 0
nlife t1 start
tsn
nlife t1 resume
tsn
nlife t1 resume
tsn
nlife t1 pause
tsn
nlife t1 pause
tsn
nlife t1 resume
tsn
