# 4BI：Dialogs

# ---- 0) 最小 load（prel.zip 默认内容里的 spark 必须脚本化）+ 带 objects 的索引 ----
imp builtin_data/launch/strings.json w:VERSION_NAME
imp builtin_data/launch/_index.json a
imp data/spark.obj.json5 spk
imp data/data.index.json5 idg
imp data/fg1.json5 ftr
imgset data/head1.png o 4 w n 32 h n 32 scale n 1 src s "data/head1.png"
lznew z1 prel.zip
lzadd z1 strings.json w:W1
lzadd z1 data/data.index.json5 idg
loadobj z1
load z:z1
dcnt

# ---- 1) 节点：根 / 头像 / 正文 / 说话人 / Dialogs 本体 ----
nmk drt - o 5 id s "drt" pos a 3 n 0 n 0 n 0 size a 3 n 200 n 100 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1
nmk dh1 drt o 5 id s "dh1" pos a 3 n 0 n 0 n 0 size a 3 n 32 n 32 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1
nmk dt1 drt o 5 id s "dt1" pos a 3 n 0 n 0 n 0 size a 3 n 4 n 4 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1
nmk dk1 drt o 5 id s "dk1" pos a 3 n 0 n 0 n 0 size a 3 n 4 n 4 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1
nadd drt dh1
nadd drt dt1
nadd drt dk1
nmk dlg1 drt o 6 id s "dlg1" pos a 3 n 0 n 0 n 0 size a 3 n 10 n 10 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 4 cls s "Dialogs" id s "dg1c" args a 0 properties o 3 head_node s "id:dh1" text s "id:dt1" talker s "id:dk1"
nadd drt dlg1
dlg dlg1
nrd dlg1 scale
nlife dlg1 start
dlg dlg1
nrd dt1 text
nrd dk1 text

# 无 fighter：正文写上、说话人清空
dialpush a 1 o 1 i18n s "HELLO"
dlg dlg1
nrd dt1 text
nrd dk1 text

# 有 fighter（命中）：说话人 = base.name（头像台面盲区，见 DESIGN）
dialnext
dialpush a 1 o 2 i18n s "HI" fighter s "fg1"
dlg dlg1
nrd dt1 text
nrd dk1 text

# 从"说话人非空"的状态隐藏：正文与说话人一起清；缩放到 0 的过程里还可见
dialclear
dlg dlg1
nrd dt1 text
nrd dk1 text
cupd dlg1 10
dlg dlg1
nrd dlg1 scale
cupd dlg1 1000
dlg dlg1
nrd dlg1 scale

# 再显示：visible 回真 + 缩放回 1
dialnext
dialpush a 1 o 2 i18n s "BACK" fighter s "fg1"
dlg dlg1
nrd dlg1 scale
cupd dlg1 1000
dlg dlg1
nrd dlg1 scale

# fighter 找不到：说话人清空
dialclear
dlg dlg1
dialnext
dialpush a 1 o 2 i18n s "GHOST" fighter s "nobody"
dlg dlg1
nrd dt1 text
nrd dk1 text
nrd dh1 image

# stop 后不再响应事件（补一次 clear + push）
nlife dlg1 stop
dialclear
dlg dlg1
nrd dt1 text
dialpush a 1 o 2 i18n s "AFTER" fighter s "fg1"
dlg dlg1
nrd dt1 text
cupd dlg1 1000
dlg dlg1

# props 指向查不到的节点：引用空值（load 跟着报 node got null，不炸）
nmk dlg2 drt o 6 id s "dlg2" pos a 3 n 0 n 0 n 0 size a 3 n 10 n 10 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 4 cls s "Dialogs" id s "dg2c" args a 0 properties o 3 head_node s "id:nope" text s "id:nope" talker s "id:nope"
nadd drt dlg2
dlg dlg2
nlife dlg2 start
dialpush a 1 o 2 i18n s "TWO" fighter s "fg1"
dlg dlg2
dialclear
dlg dlg2
