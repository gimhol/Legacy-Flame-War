# POINTER_*：经真分发器（台面 op `cmd`）在 `lfw.ui_layers()` 上重放命中/派发
# 页 pg1 = 根(组件 p) + ch1(组件 1) + ch2(组件 2, visible 0) + ch3(组件 3, disabled 1) + ch4(组件 4, stop_click)
# `flpage <别名> <层> <页| -1=栈顶> [子下标…]` 把命令作用的那一栈的节点挂给 `nrd` 读。
uipg pg1 o 6 pos a 3 n 0 n 0 n 0 size a 3 n 8 n 8 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 3 cls s "FakeComp" id s "p1c" properties o 2 n n 1 s s "p" items a 4 o 6 id s "ch1" pos a 3 n 0 n 0 n 0 size a 3 n 1 n 1 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 3 cls s "FakeComp" id s "c1" properties o 2 n n 1 s s "1" o 7 id s "ch2" visible n 0 pos a 3 n 0 n 0 n 0 size a 3 n 1 n 1 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 3 cls s "FakeComp" id s "c2" properties o 2 n n 1 s s "2" o 7 id s "ch3" disabled n 1 pos a 3 n 0 n 0 n 0 size a 3 n 1 n 1 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 3 cls s "FakeComp" id s "c3" properties o 2 n n 1 s s "3" o 6 id s "ch4" pos a 3 n 0 n 0 n 0 size a 3 n 1 n 1 n 1 center a 3 n 0 n 0 n 0 scale a 3 n 1 n 1 n 1 component a 1 o 3 cls s "FakeComp" id s "c4" properties o 3 n n 1 s s "4" stop_click b 1

# 校验告警：层找不到 / 页找不到 / 页变了（下标处 id 与 --pages 不符）/ 节点找不到
cmd s "POINTER_DOWN --paths=7,0 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_DOWN --paths=0,5 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_DOWN --paths=0,0 --pages=nope --points=1,2,3 --b=0"
cmd s "POINTER_DOWN --paths=0,0,9 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_DOWN --paths=0,0 --pages=pg1 --points=1,2 --b=0"
cmd s "POINTER_DOWN --paths=0,0 --pages=pg1 --points=1,2,3 --b=x"
# DOWN：近到远 = 列表次序（ch1 再根）；ch3 被 disabled 挡掉
cmd s "POINTER_DOWN --paths=0,0,2|0,0,0|0,0 --pages=pg1|pg1|pg1 --points=1,2,3|4,5,6|7,8,9 --b=0"
nrd c1 ptr
nrd top ptr
# MOVE：enter → stay（第二次不再 enter）→ 路径消失 = 全体 leave
cmd s "POINTER_MOVE --paths=0,0,0 --pages=pg1 --points=1,2,3 --b=0"
nrd c1 ptr
cmd s "POINTER_MOVE --paths=0,0,0 --pages=pg1 --points=2,2,3 --b=0"
nrd c1 ptr
cmd s "POINTER_MOVE --b=0"
nrd c1 ptr
# UP：同一目标 ⇒ `pointer_down` 与 `click_flag` 两个分支都走（up + click）
cmd s "POINTER_DOWN --paths=0,0,0 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_UP --paths=0,0,0 --pages=pg1 --points=1,2,3 --b=0"
nrd c1 ptr
# MOVE 的 leave 会清 `click_flag` ⇒ 之后的 UP 只 up 不 click
cmd s "POINTER_DOWN --paths=0,0,0 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_MOVE --paths=0,0,0 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_MOVE --b=0"
cmd s "POINTER_UP --paths=0,0,0 --pages=pg1 --points=1,2,3 --b=0"
nrd c1 ptr
# UP：目标里没有仍按下的 ch1 ⇒ 走「取消未收到 up 的按下节点」尾巴
cmd s "POINTER_DOWN --paths=0,0,0 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_UP --paths=0,0,1 --pages=pg1 --points=1,2,3 --b=0"
nrd c1 ptr
# CANCEL：取消所有仍按下的节点（空表再跑一次 ⇒ 无动作）
cmd s "POINTER_DOWN --paths=0,0 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_CANCEL --b=2"
nrd top ptr
cmd s "POINTER_CANCEL --b=0"
# CLICK：只派发 click（禁用项被跳过）；`stop_click` 的组件让「近到远」当场中断
cmd s "POINTER_CLICK --paths=0,0,2 --pages=pg1 --points=1,2,3 --b=0"
cmd s "POINTER_CLICK --paths=0,0,0|0,0 --pages=pg1|pg1 --points=1,2,3|7,8,9 --b=2"
cmd s "POINTER_CLICK --paths=0,0,3|0,0,0 --pages=pg1|pg1 --points=1,2,3|4,5,6 --b=0"
# ENTER / LEAVE：直接派发（visible 0 的项被跳过；无 --paths 的 LEAVE 让所有悬停项 leave）
cmd s "POINTER_ENTER --paths=0,0,0 --pages=pg1"
cmd s "POINTER_ENTER --paths=0,0,1 --pages=pg1"
nrd c1 ptr
cmd s "POINTER_LEAVE --paths=0,0,0 --pages=pg1"
cmd s "POINTER_ENTER --paths=0,0 --pages=pg1"
cmd s "POINTER_LEAVE"
nrd top ptr
# 光标表（平台层画光标用）：只有给了 --from 的发起方才会被写
ptcur z

ptcur z

ptcur z
cmd s "POINTER_UP --from=z --paths=0,0,0 --pages=pg1 --points=3,4,5 --b=0"
ptcur z
cmd s "POINTER_LEAVE --from=z"
ptcur z
# CANCEL 的光标面：`down=false`、`hidden` 保留原值（`nullopt` 分支）
cmd s "POINTER_CANCEL --from=z --b=0"
ptcur z
ptcur -
