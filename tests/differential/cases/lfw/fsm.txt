# ---- 1) use 语义：同状态也 leave+enter；未知键 → set_state(null) ----
gfuse rt c1 Running

gfr rt c1
gfu rt c1 50

gfr rt c1

# ---- 2) Running：暂停门（暂停时 update 只走时钟）----
gfuse rt c1 Running
wpause 1
gfu rt c1 100
gfr rt c1
wpause 0
gfu rt c1 100
gfr rt c1

# ---- 3) BeforeEnd：3000 门槛 + 暂停门 ----
gfu rt c1 1000
gfr rt c1
wpause 1
gfu rt c1 2000
gfr rt c1
wpause 0
gfu rt c1 1999
gfr rt c1
gfu rt c1 1
gfr rt c1
nrd sb flags

# ---- 4) End：leave 隐藏 score_board；再 use Running 会走 leave ----
gfuse rt c1 Running
nrd sb flags
gfr rt c1

# ---- 5) 重进 BeforeEnd：enter 清零（用 2999/3000 边界钉住）----
gfuse rt c1 BeforeEnd
gfu rt c1 2999
gfr rt c1
gfu rt c1 1
gfr rt c1
nrd sb flags

# ---- 6) End 再进一次：preset 是 end（win 在台面不可达）----
gfuse rt c1 End
gfr rt c1
nrd sb flags
gfuse rt c1 Running
gfr rt c1

# ---- 7) End → BeforeEnd 路径（上一个是 End，leave 收尾）----

gfuse rt c1 BeforeEnd

gfu rt c1 2999
gfr rt c1
gfu rt c1 1
gfr rt c1
