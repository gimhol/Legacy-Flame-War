# 魔法 id：1000 / 999 / -999（返回共享常量对象）
nf n 1000 repeat
nf s "1000" repeat
nf n 999 repeat
nf n -999 repeat
nf s "-999" repeat
nf s "999" frame

# 0：zero_as 决定 {id:"0"} 还是 {}

nf s "0" frame
nf n -0 frame
nf s "-0" repeat
nf n 0 noop
