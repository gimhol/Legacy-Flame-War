# state_base
# `env victim` takes a string value.
# `env state|dataset|pos|velx|velz` take a value literal (`velx`/`velz` accept `u`).
# run make|leave|restrict <x> <y> <z>|update
# StateEnum.HealSelf = 1700 ; Defines.STATE_HEAL_SELF_HP = 104
# Buff_Healing.duration_of(e, hp) = ceil(hp / max(1, hp_healing_value)) * max(1, hp_healing_ticks)
# State_Base.leave only handles HealSelf (strict ===); on_restrict keeps MIN_V = 0.5 of the
# velocity on the axes that actually moved (a moved y pulls in both x and z).
env victim s "V1"
env dataset o 2 hp_healing_value n 10 hp_healing_ticks n 2
env state n 1700
run make
run leave
# ---- every other state does nothing ----
env state n 0
run make
run leave
# ---- a string state is not `===` HealSelf ----
env state s "1700"
run make
run leave
# ---- the heal amount is clamped to one ----
env dataset o 2 hp_healing_value n 0 hp_healing_ticks n 2
env state n 1700
run make
run leave
# ---- the tick count is clamped to one ----
env dataset o 2 hp_healing_value n 10 hp_healing_ticks n 0
run make
run leave
# ---- a missing dataset key poisons the duration ----
env dataset o 1 hp_healing_ticks n 2
run make
run leave
# ---- update is a no-op ----
run update
# ---- restrict: only x moved ----
env pos o 3 x n 0 y n 0 z n 0
env velx n 2
env velz n -2
run restrict 10 0 0
# ---- restrict: a moved y clamps both axes ----
env pos o 3 x n 0 y n 0 z n 0
env velx n 2
env velz n -2
run restrict 0 10 0
# ---- restrict: only z moved ----
env pos o 3 x n 0 y n 0 z n 0
env velx n 2
env velz n -2
run restrict 0 0 10
# ---- restrict: nothing moved ----
env pos o 3 x n 5 y n 6 z n 7
env velx n 2
env velz n -2
run restrict 5 6 7
# ---- restrict: a velocity inside the clamp is kept ----
env pos o 3 x n 0 y n 0 z n 0
env velx n 0.25
env velz n -0.25
run restrict 10 0 10
# ---- restrict: a positive z velocity is clamped too ----
env pos o 3 x n 0 y n 0 z n 0
env velx n 0.25
env velz n 2
run restrict 0 0 10
# ---- restrict: a missing velocity stays missing ----
env pos o 3 x n 0 y n 0 z n 0
env velx u
env velz u
run restrict 10 10 10
