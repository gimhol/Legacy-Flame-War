# character_state_jump
# `env state|pos|gy|jx|jy|jz|jt|jumpflag|dvals|onlanding|landing1` take value literals;
# `env lr|ud <bare number>`; `env bot <truthy>`; `env held s "..."` lists held keys
# (`is_end(key)` is true when the key is NOT in the string, i.e. released).
# run make|default|enter|update|landing
# StateEnum.Jump=4; GK: R/L/U/D/j
# `landing1` must stay a string when the landing fallback is exercised: C++ renders
# `to_string(undefined)` as the string "undefined" while TS logs `u`.

env state n 4
env pos o 3 x n 0 y n 0 z n 0
env gy n 0
env jx n 1
env jy n 2
env jz n 3
env jt n 4
env bot b 0
env held s ""
env lr 0
env ud 0
env jumpflag b 0
env dvals o 7 jump_height n 10 jump_h_f n 3 jump_distancez n 5 jump_z_f n 7 jump_distance n 11 jump_x_f n 13 atom_time n 1
env onlanding u
env landing1 s "L1"
run make
run default
# ---- enter zeroes the four jumping fields ----
run enter
# ---- in the air only the decay runs ----
env pos o 3 x n 0 y n 1 z n 0
run update
# ---- the ground check follows the actual ground height ----
env gy n 2
env pos o 3 x n 0 y n 2 z n 0
env jx n 0
env jy n 0
env jz n 0
env jt n 0
run update
env pos o 3 x n 0 y n 3 z n 0
run update
env gy n 0
# ---- on the ground only t advances when no key is held ----
env pos o 3 x n 0 y n 0 z n 0
env jx n 0
env jy n 0
env jz n 0
env jt n 0
run update
# ---- R held advances +x ----
env held s "R"
env jx n 0
env jy n 0
env jz n 0
env jt n 0
run update
# ---- L held advances -x ----
env held s "L"
env jx n 0
run update
# ---- U held advances -z ----
env held s "U"
env jz n 0
run update
# ---- D held advances +z ----
env held s "D"
env jz n 0
run update
# ---- j held advances +y ----
env held s "j"
env jy n 0
run update
# ---- all keys held: the two x steps cancel, so do the two z steps ----
env held s "RLUDj"
env jx n 0
env jy n 0
env jz n 0
env jt n 0
run update
# ---- a fractional atom time is rounded ----
env held s ""
env dvals o 7 jump_height n 10 jump_h_f n 3 jump_distancez n 5 jump_z_f n 7 jump_distance n 11 jump_x_f n 13 atom_time n 1.2345
env jx n 0
env jy n 0
env jz n 0
env jt n 0
run update
# ---- the bot branch only advances t and y ----
env bot b 1
run update
env bot b 0
# ---- the jump flag launches the jump ----
env dvals o 7 jump_height n 10 jump_h_f n 3 jump_distancez n 5 jump_z_f n 7 jump_distance n 11 jump_x_f n 13 atom_time n 1
env jumpflag b 1
env lr 1
env ud -1
env jt n 2
env jy n 1
env jx n 0
env jz n 0
run update
# ---- t = 0 clamps to the minimum ----
env jt n 0
run update
# ---- lr / ud flip the signs ----
env lr -1
env ud 1
env jt n 2
run update
# ---- a truthy string jump flag still launches ----
env jumpflag s "0"
run update
# ---- a falsy jump flag skips the launch ----
env jumpflag n 0
run update
# ---- with a zero atom time the jump time stays 0, so the minimum applies ----
env jumpflag b 1
env dvals o 7 jump_height n 10 jump_h_f n 3 jump_distancez n 5 jump_z_f n 7 jump_distance n 11 jump_x_f n 13 atom_time n 0
env jt n 0
env jy n 2
env lr 1
env ud -1
run update
# ---- landing: the frame wins ----
env onlanding o 1 id s "55"
run landing
env onlanding u
# ---- landing: landing_1 plus update_velocity ----
env landing1 s "L1"
run landing
# ---- the landing frame id is re-read ----
env landing1 s "L9"
run landing
