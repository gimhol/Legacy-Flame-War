# buff_marks
# `env id|kind|cls|victim` take a string value; `env vpos|vframe|mark` take an object.
# Buff_GroupAttack: effect_oid="fx" effect_frame_id="16" mark kind "GroupAttack"
# Buff_Electrify:   effect_oid="fx" effect_frame_id="32" mark kind "Electrify"
env cls s "group_attack"
env kind s "GroupAttack"
env id s "B1"
env victim s "V1"
env vpos o 3 x n 10 y n 20 z n 30
env vframe o 3 centery n 4 height n 8 pic_h n 0
# ---- mount marks the victim and registers the buff in the world ----
run make
run mount
# a second mount is a no-op
run mount
# ---- the effect entity is placed at the victim frame centre ----
run effect
# the effect is re-placed on every update but created only once
run effect
# ---- unmount removes the mark and drops the victim ----
run unmount
# a second unmount is a no-op
run unmount
# ---- a mark carrying a foreign value must not be deleted ----
env victim s "V2"
env vpos o 3 x n 1 y n 2 z n 3
env vframe o 3 centery n 5 height n 0 pic_h n 12
run make
run mount
env mark o 2 key s "GroupAttack" value s "FOREIGN"
run unmount
# ---- electrify uses the same shape with another frame id ----
env cls s "electrify"
env kind s "Electrify"
env id s "B2"
env victim s "V3"
env vpos o 3 x n 0 y n 0 z n 0
env vframe o 3 centery n 0 height n 0 pic_h n 0
run make
run mount
run effect
env mark o 2 key s "Electrify" value s "FOREIGN"
run unmount
env victim s "V4"
env vpos o 3 x n 7 y n 8 z n 9
env vframe o 3 centery n 1 height n 2 pic_h n 3
env cls s "group_attack"
env kind s "GroupAttack"
env id s "B3"
run make
run effect
run mount
run unmount
