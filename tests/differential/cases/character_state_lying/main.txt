# character_state_lying
# `env state` is the constructor argument, `env estate` is the entity's own state.
# Other env keys: `pos|gy|frameid|hp|hpr|hpmax|holdtype|holdteam|team|tough|tmax|trest|la|ld|lc|wait|dvals|deadjoin|deadgone|reserve|wakeup|motionless|invul|blink|outline|puppets`
# take value literals; `env holding <truthy>`; `env held s "..."` lists held keys
# (`is_end(key)` is true when the key is NOT in the string, i.e. released).
# run make|default|enter|update|leave|dead|findframe
# StateEnum.Lying=14; WeaponEnum.Heavy=2; GK.a="a", GK.d="d"
# dvals keys: atom_time / gone_blink_time / lying_blink_time
# The fakes store raw values: they skip the real Entity's normalisation
# (`round_float(max(0, v))` on blinking / invulnerable, `round_float` on reserve /
# toughness) so both harnesses stay identical.

env state n 14
env pos o 3 x n 1 y n 0 z n 2
env gy n 0
env frameid s "F0"
env hp n 5
env hpr n 4
env hpmax n 9
env holding b 0
env holdtype n 2
env holdteam n 0
env team s "1"
env tough n 3
env tmax n 7
env trest n 2
env la n 0
env ld n 0
env lc n 0
env wait n 4
env held s ""
env dvals o 3 atom_time n 1 gone_blink_time n 30 lying_blink_time n 20
env deadjoin u
env deadgone n 0
env reserve n 0
env wakeup n 0
env motionless n 0
env invul n 0
env blink n 0
env outline s "old"
env puppets u
run make
env state n 3
run make
run default
env state n 14
# ---- enter zeroes the three counters, restores the toughness and drops the holding ----
run enter
# ---- a light weapon is dropped but keeps its team ----
env holding b 1
env holdtype n 4
run enter
# ---- an alive entity never reports a death ----
env hp n 5
env reserve n 2
env team s "1"
env puppets a 1 s "1"
run enter
# ---- a heavy weapon is handed to the lying entity's team ----
env holdtype n 2
run enter
# ---- an unknown holding type is dropped without a team change ----
env holdtype u
run enter
env holding b 0
# ---- nothing happens while hp is above zero ----
env hp n 5
run enter
# ---- dead with no reserve, no join and no gone flag ----
env hp n 0
env reserve n 0
run enter
# ---- dead, reserve left and the team has a puppet: blink and respawn ----
env reserve n 2
env team s "1"
env puppets a 1 s "1"
run enter
# ---- the reserve runs out, so nothing respawns ----
env reserve n 1
env puppets a 1 s "2"
run enter
# ---- a pending join swallows the death ----
env reserve n 1
env puppets a 1 s "1"
env deadjoin o 1 hp n 5
run enter
# ---- with reserve gone and no join, the gone flag blinks the entity away ----
env reserve n 1
env puppets a 1 s "1"
env deadjoin z
env deadgone n 1
run enter
# ---- the team is searched among all puppets ----
env reserve n 2
env deadjoin u
env deadgone n 0
env team s "2"
env puppets a 2 s "1" s "2"
run enter
# ---- no puppets at all, so the team is never found ----
env reserve n 2
env puppets a 0
run enter
# ---- a string reserve still counts down numerically ----
env reserve s "3"
env puppets a 1 s "1"
run enter
# ---- on_dead can be called on its own ----
env reserve n 3
env hp u
run dead
env reserve n 2
env team s "1"
env puppets a 1 s "1"
run dead
# ---- the last reserve is spent, so nothing respawns ----
env reserve n 1
run enter
run dead
# ---- a pending join swallows the death before the gone flag ----
env reserve n 0
env deadjoin o 1 hp n 5
env deadgone n 1
run dead
env deadjoin u
env deadgone n 0
env hp n 0
# ---- update: both keys released counts nothing ----
env held s ""
env la n 0
env ld n 0
env lc n 0
env wait n 4
run update
# ---- attack held on an odd count pulls the wait down and returns early ----
env held s "a"
env la n 1
env ld n 2
env lc n 0
env wait n 4
run update
# ---- the next attack count is even, so the attack branch is skipped ----
run update
# ---- an even attack count falls through to the defend branch ----
env held s "ad"
env la n 2
env ld n 1
env lc n 0
env wait n 4
run update
# ---- defend held on an odd count pulls the wait up ----
env held s "d"
env la n 2
env ld n 1
env lc n 0
env wait n 4
run update
# ---- a released defend key never pulls the wait up ----
env held s "a"
env la n 2
env ld n 3
env lc n 0
env wait n 4
run update
# ---- a negative defend count is truthy too ----
env held s "d"
env la n 2
env ld n -3
env lc n 0
env wait n 4
run update
# ---- the defend wait is rounded ----
env dvals o 3 atom_time n 1.2345 gone_blink_time n 30 lying_blink_time n 20

run update

# ---- a released attack key never pulls the wait down ----
env held s "d"

env ld n 2

env wait n 4
run update
# ---- an even defend count is skipped ----
env held s "d"
env la n 2

run update
# ---- a negative odd count is truthy, so the attack branch still runs ----
env held s "a"
env la n -3
env ld n 2
env wait n 4
run update

env la n 1.5
run update
# ---- a string count is concatenated, and its modulo is numeric ----

run update

run update

env wait u
run update
# ---- a NaN defend count is skipped ----
env held s "d"
env la n 2
env ld u
env wait n 4
run update
# ---- an odd defend count still pulls the wait up ----
env ld n 3
run update
# ---- leave without a join only refreshes the blink counters ----

env hp n 0

env hp n 0

env wakeup n 0
run leave

env hp n 1

env blink n 8
run leave

env deadjoin z

run leave
