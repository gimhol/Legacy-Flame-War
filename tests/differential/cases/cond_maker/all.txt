# 基本串联（运算符用裸 token，不加引号）
mk a
add a n 1 == n 1
done a
add a n 2 == n 2
done a
mk b
add b n 1 == n 1
or b n 2 == n 2
or b n 3 == n 3
done b
mk c
add c n 1 > n 0
and c s "x" == s "x"
done c
mk d
add d n 1 != n 2
and d n 3 <= n 4
and d n 5 >= n 6
or d n 7 < n 8
done d

# 括号 / wrap / not / 嵌套
mk w
wrap w n 1 == n 1
done w
not w n 1 == n 1
done w
mk nst
nested nst n 1 == n 1 n 2 == n 2
done nst
mk nst2
or nst2 n 1 == n 1
nested nst2 n 2 == n 2 n 3 == n 3
done nst2
mk nst3
not nst3 n 1 == n 1
or nst3 n 2 == n 2
done nst3

# one_of / not_in 家族
mk o1
oneof o1 n 7 n 1 n 2 n 3
done o1
mk o2
add o2 n 1 == n 1
andoneof o2 n 7 n 1 n 2
done o2
mk o3
oroneof o3 n 7 n 1 n 2
done o3
mk o4
notin o4 n 7 n 1 n 2
done o4
mk o5
andnotin o5 n 7 n 1 n 2
done o5
mk o6
ornotin o6 n 7 n 1 n 2
done o6
mk o7
oneof o7 s "k" s "a" s "b"
done o7

# JS ToString：布尔、1e21、-0
mk t1
add t1 b 1 == b 0
done t1
add t1 b 1 == b 1
done t1
mk t2
add t2 n 1000000000000000000000 == n 1
done t2
mk t3
add t3 n -0 == n 0
done t3
mk t4
add t4 s "1" == n 1
done t4

# 引号
mk q1
quote q1 b1 dq
add q1 s "he\"llo" == s "it's"
done q1
mk q2
quote q2 b1 sq
add q2 s "he\"llo" == s "it's"
done q2
mk q3
quote q3 b0 dq
add q3 s " a " == s "b"
done q3
mk q4
quote q4 b1 bad
done q4
mk q5
quote q5 b1 dq
done q5

# 空白裁剪（NBSP / LS / PS / 前后空格）
mk s1
add s1 s "\u00a0x\u00a0" == s "y"
done s1
mk s2
add s2 s "\u2028x" == s "y\u2029"
done s2
mk s3
add s3 s "  x  " == s "  y  "
done s3
mk s4
add s4 s "\t\nx\ry" == s "z"
done s4

# term_format 自定义
mk f1
term f1 wrap
add f1 n 1 == n 2
or f1 n 3 == n 4
done f1
mk f2
term f1 wrap
or f2 n 1 == n 2
done f2

# 错误：缺少运算符
mk e1
add e1 n 1 == n 1
add e1 n 2 == n 2
done e1

# 错误：操作数为 nullish
mk e2
add e2 u == n 1
done e2
mk e3
add e3 z == n 1
done e3
mk e4
add e4 n 1 == u
done e4

# 错误：非法操作数（对象 / 数组）
mk e5
addobj e5 == n 1
done e5
mk e6
addarr e6 == n 1
done e6

# 错误：非法与保留运算符
mk e7
addtag e7 empty n 1 n 2
done e7
mk e8
addtag e8 or n 1 n 2
done e8
mk e9
addtag e9 and n 1 n 2
done e9
mk e10
addtag e10 not n 1 n 2
done e10

# 错误：one_of / not_in 无值
mk e11
oneof0 e11 n 1
done e11
mk e12
notin0 e12 n 1
done e12

# 错误：not 跟在项后
mk e13
add e13 n 1 == n 1
not e13 n 2 == n 2
done e13

# 报错后的后续操作应保持"已报错"；重新 mk 后恢复可用
add e13 n 5 == n 5
done e13
mk e1
done e1

# 子构造器继承引号设置
mk oq
quote oq b1 dq
oneof oq s "k" s "a" s "b"
done oq

# 子构造器里的错误要向上传播（用 nullish 操作数触发，空运算符无法用裸 token 表达）
mk pe
not pe n 1 == u
done pe
mk pf
wrap pf u == n 1
done pf

# 中段片段的前后空白也必须被裁剪（末尾 trim 覆盖不到）
mk mid1
add mid1 n 1 == n 1
and mid1 s " a" == s "b"
done mid1
mk mid2
add mid2 n 1 == s "x "
and mid2 n 2 == n 3
done mid2
