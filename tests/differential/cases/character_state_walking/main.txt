# character_state_walking
# `env victim` takes a string value;
# `env ctrlud|ctrllr|hweapon` take a bare number (0/1).
# `env state|hp|vwait|frame|ground_y|holding|indexes|pos` take a value literal (`vwait` accepts u).
# run make|update
# StateEnum.Walking=1 ; WeaponEnum.Heavy=2 ; is_weapon(holding) == holding.data.type === 16
# The entity side supplies `get_sudden_death_frame` as {id:"SD"} and logs the call.
env victim s "V1"
env state n 1
env hp n 10
env vwait u
env waitflag 4
env frame o 1 wait n 7
env ground_y n 0
env holding n 0
env hweapon 0
env ctrlud 0
env ctrllr 0
env indexes o 2 default s "50" in_the_skys a 2 s "70" s "71"
env pos o 3 x n 1 y n 0 z n 2
run make
# ---- idle: no input, no wait -> the default frame is entered with fallback ----
run update
# ---- holding up suppresses the idle branch ----
env ctrlud 1
run update
env ctrlud 0
# ---- holding left/right suppresses the idle branch ----
env ctrllr 1
run update
env ctrllr 0
# ---- a truthy wait suppresses the idle branch ----
env vwait n 3
run update
# ---- a zero wait does not suppress it ----
env vwait n 0
run update
# ---- a heavy weapon in hand refreshes the wait instead of the frame ----
env vwait u
env hweapon 1
env holding n 2
run update
# ---- a weapon that is not heavy enters the default frame ----
env vwait u
env holding n 1
run update
# ---- a heavy holding that is not a weapon enters the default frame ----
env vwait u
env hweapon 0
env holding n 2
run update
# ---- a string "2" holding is not === Heavy ----
env vwait u
env hweapon 1
env holding s "2"
run update
env hweapon 0
env holding n 0
# ---- a dead entity enters the sudden death frame and skips the rest ----
env hp n 0
env pos o 3 x n 1 y n 5 z n 2
env ground_y n 4
run update
# ---- above the ground enters the sky frame ----
env hp n 10
run update
# ---- exactly on the ground does nothing more ----
env pos o 3 x n 1 y n 4 z n 2
run update
# ---- a missing hp counts as alive ----
env hp u
run update
