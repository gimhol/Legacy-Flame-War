# character_state_dash
# `env victim` takes a string value; `env ctrlud|ctrllr` take a bare number (-1/0/1).
# `env state|prevstate|pos|ground_y|velx|vely|velz|facing|dataset` take a value literal.
# run make|enter
# StateEnum.Running=2 ; dx = dash_distance * dash_x_f ; dz = dash_distancez * dash_z_f
# vy = dash_height * dash_h_f
env victim s "V1"
env state n 5
env prevstate n 0
env pos o 3 x n 0 y n 0 z n 0
env ground_y n 0
env velx n 0
env vely n 0
env velz n 9
env facing n 1
env ctrlud 0
env ctrllr 0
env dataset o 6 dash_distance n 10 dash_x_f n 0.5 dash_distancez n 4 dash_z_f n -2 dash_height n 3 dash_h_f n 1
run make
# ---- every branch of the x selection ----
# ---- on the ground: the previous running state faces forward ----
env prevstate n 2
env facing n -1
env velx n 5
run enter
# ---- a string "2" previous state is not === Running ----
env prevstate s "2"
env velx n 5
run enter
# ---- every other previous state falls back to the velocity ----
env prevstate n 1
env velx n 5
run enter
# ---- left/right input wins over the current velocity ----
env prevstate n 0
env ctrllr 1
env velx n 0
run enter
# ---- a negative left/right input mirrors the distance ----
env ctrllr -1
run enter
# ---- a positive x velocity dashes forward ----
env ctrllr 0
env velx n 5
run enter
# ---- a negative x velocity dashes backward ----
env velx n -5
run enter
# ---- a negative x velocity with a forward facing still dashes backward ----
env facing n 1
run enter
# ---- a zero x velocity dashes facing forward ----
env velx n 0
env facing n 1
run enter
# ---- a zero x velocity with a backward facing dashes backward ----
env velx n 0
env facing n -1
run enter
# ---- a missing x velocity dashes facing forward ----
env velx u
run enter
# ---- up/down input writes z from the z distance ----
env velx n 0
env ctrlud 1
run enter
# ---- the opposite up/down mirrors the z distance ----
env ctrlud -1
run enter
# ---- airborne with a vertical speed aborts ----
env ctrlud 0
env pos o 3 x n 0 y n 5 z n 0
env ground_y n 4
env vely n 3
run enter
# ---- airborne without a vertical speed dashes ----
env vely n 0
run enter
# ---- exactly on the ground dashes even with a vertical speed ----
env pos o 3 x n 0 y n 4 z n 0
env vely n 3
run enter
# ---- a missing vertical speed dashes ----
env vely u
run enter
# ---- a missing dataset poisons the distances ----
env dataset o 0
run enter
