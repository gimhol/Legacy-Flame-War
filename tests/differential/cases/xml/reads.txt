# get_* 回落链：先同名子元素，再同名属性
new gp parent
attr gp g s "attr-g"
attr gp ga s "attr-ga"
attr gp gn n 7
attr gp gb s "false"
rd gp getstr g
rd gp getnum gn
rd gp getbool gb
rd gp getstr missing
rd gp getstr_or missing "def"
rd gp getnum_or missing 9
rd gp getbool_or missing 1
rd gp getstr_or g "def"
rd gp getnum_or gn 9
rd gp getbool_or gb 1

new gc g
attr gc type s "string"
attr gc value s "child-g"
ins gp gc -
rd gp getstr g
rd gp getstr_or g "def"

new gaS ga
attr gaS type s "string"
attr gaS value s "child-ga"
ins gp gaS -
rd gp getstr ga

# as_object：属性全进（含 type），子带 name 用 name 作键
new ao obj
attr ao type s "object"
attr ao extra s "e"
new aoc1 leaf
attr aoc1 type s "string"
attr aoc1 value s "v1"
attr aoc1 name s "k1"
ins ao aoc1 -
new aoc2 leaf
attr aoc2 type s "string"
attr aoc2 value s "v2"
ins ao aoc2 -
new aoc3 leaf
attr aoc3 type s "string"
attr aoc3 value s "v3"
attr aoc3 name s ""
ins ao aoc3 -
new aoc4 item
attr aoc4 type s "array"
new aoc4i i
attr aoc4i type s "number"
text aoc4i "8"
ins aoc4 aoc4i -
attr aoc4 name s "k4"
ins ao aoc4 -
attr aoc4 name s "k4"
rd ao asobj
rd ao asval
rd ao getobj obj
rd ao getobj missing
rd ao getobj leaf
