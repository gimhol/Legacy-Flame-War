# stringify 怪癖：同名子元素分组、空元素被丢、转义、子存在时文本被丢、属性保序
new g root
new g1 a
attr g1 x s "1"
new g2 b
attr g2 y s "2"
new g3 a
attr g3 x s "3"
ins g g1 -
ins g g2 -
ins g g3 -
rd g children
dump g
rd g strop

# 空元素在 stringify 里被丢（父的孩子表里还在）
new drop p
new de e
ins drop de -
rd drop children
dump drop
rd drop strop

# 空元素的空元素
new d2 p
new d3 c
new d4 d
ins d3 d4 -
ins d2 d3 -
dump d2
rd d2 strop

# 空元素排在同名有内容孩子后面（array 里的 undefined 项）
new a1 arr
new ae e
ins a1 ae -
new ar r
attr ar v s "1"
ins a1 ar -
new ae2 e
ins a1 ae2 -
rd a1 children
rd a1 strop

# 子存在时文本被丢
new tx p
text tx "hello"
rd tx strop
new tc c
ins tx tc -
rd tx strop
dump tx

# 转义
new esc e
attr esc amp s "&<>\"'"

text esc "&<>\"'"
rd esc strop
new esc2 e
text esc2 "tab\there"
rd esc2 strop
new esc3 e
attr esc3 empty s ""
rd esc3 strop
rd esc3 attrs

# set_attr：nullish 删属性；数组 join
attr ord e u
rd ord attrs

rd ord attrs

rd ord attr arr

# sattr：尾部 nullish 丢、中间变空、全空删、非数组直落 set_attr

rd ord attr sa

rd ord attrs
