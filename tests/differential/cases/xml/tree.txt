# 建树 / 改树：create + attr + ins（含 index、重复挂）+ dump / stringify / children / cbt
new r root
attr r ver s "1.0"
attr r n n 2

new c1 a
attr c1 v s "one"
ins r c1 -

new c2 b
attr c2 v s "two"
ins r c2 -

new c3 a
attr c3 v s "three"
ins r c3 0

rd r children
dump r
rd r strop

# insert 到中间
new c4 m
ins r c4 1
rd r children
dump r

# 同一个孩子挂到两处（tool 语义：不先摘旧的）
new p1 pa
new p2 pb
ins p1 c4 -
ins p2 c4 -
rd p1 children
rd p2 children
rd c4 parent

# remove
rm r c3
rd r children
rm r c3
rm p1 c4
rd p1 children
rd c4 parent
ins p1 c4 -
rd c4 parent
rmall p1
rd c4 parent
rd c4 cbt nope
rd c4 tag

# remove_self / remove_all
new solo x
rmself solo
ins r solo -
rmself solo
rd r children
rmall r
rd r children
dump r
rd r strop
rmall p2
rd c4 parent

# 空树各种形态
new e1 item
rd e1 strop
new e2 item
new e3 child
ins e2 e3 -
rd e2 strop
dump e2

# 文本树
new t1 text
text t1 "l1\nl2"
rd t1 text
rd t1 strop
new t2 text
text t2 "  pad  "
attr t2 x s "1"
rd t2 strop
