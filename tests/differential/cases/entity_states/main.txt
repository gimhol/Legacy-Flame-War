# entity_states
# `env type <value literal>` + `env code <bare number>` feed `run fallback`.
# `run size | dump | head | tail | get <key literal> | fallback | fallback2 | setdup`
# StateEnum: Standing=0, Falling=12, LandGoto94=100, Weapon_OnGround=1004,
#            HeavyWeapon_JustOnGround=2002, Ball_Flying=3000, TransformTo 8001..8999
# EntityEnum: Fighter=8, Weapon=16, Ball=32
# The dump prints every registry entry in insertion order, so the whole wiring is
# compared entry by entry.

run size
run head
run tail
run dump
# ---- get: numeric keys survive the round trip ----
run get n 0
run get n -1
run get n -2
run get n -3
run get n 8001
run get n 8999
run get n 9000
run get n 3000
run get n 100
run get n 1004
run get n 2002
run get n 1
run get n 11
# ---- get: string keys are distinct from numeric ones ----
run get s "0"
run get s "8001"
run get s "-1"
run get s "8_23"
run get z
run get b 1
run get u
# ---- has: the same key shapes ----
run has n 0
run has s "0"
run has n 8999
run has n 9000
run has s "8001"
run has z
run has u
run has n 3000
run has s "8_23"
# ---- the default fallback branch builds a string key and caches it ----
env type n 23
env code 7
run fallback
run fallback2
run size
run get s "23_7"
# ---- the fighter fallback branch ----
env type n 8
env code 23
run fallback
run fallback2
run get s "8_23"
run size
# ---- the weapon fallback branch ----
env type n 16
env code 23
run fallback
run get s "16_23"
# ---- the ball fallback branch ----
env type n 32
env code 23
run fallback
run get s "32_23"
# ---- an unknown entity type falls back to the bare state ----
env type n 99
env code 23
run fallback
run get s "99_23"
# ---- an undefined type still produces its own string key ----
env type u
env code 23
run fallback
run get s "undefined_23"
run size
# ---- a string type is not the numeric type, but shares the key ----
env type s "8"
env code 23
run fallback
run size
# ---- a fallback key never shadows the numeric state key ----
env type n 8
env code 0
run fallback
run get n 0
run get s "8_0"
run size
# ---- re-setting a key keeps its position in the table ----
run get n 0
run setdup
run get n 0
run size
run head
run tail
run get n 8999
