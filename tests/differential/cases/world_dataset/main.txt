# WorldDataset: defaults, tracked fields, hooks and dumps

# --- a non-pure dataset: every field has a getter/setter, so `Object.keys` is blind to them
run make
run keys
run dump
run has s "jump_height"
run has s "__is_world_dataset__"
run has s "nope"
run tracked s "jump_height"
run tracked s "__is_world_dataset__"
run tracked s "nope"
run get s "jump_height"
run get s "gravity"
run get s "screen_w"
run get s "screen_h"
run get s "difficulty"
run get s "sync_render"
run get s "__is_world_dataset__"
run get s "nope"
run set s "__is_world_dataset__" b 0
run get s "__is_world_dataset__"

# --- tracked writes fire the field hook and then the dataset change callback
env hook s "jump_height"
env pure b 0
run make
run set s "jump_height" n 24.5
run get s "jump_height"
run set s "jump_height" n 24.5
run get s "jump_height"
run set s "jump_height" n 30
run set s "jump_height" u
run set s "jump_height" u
run set s "jump_height" n 0
run set s "jump_height" n -0
run get s "jump_height"

# --- the callbacks span every tracked key, not just the hooked one
run set s "gravity" n 0.5
run set s "screen_w" n 800
run set s "screen_h" n 600

# --- a key outside the table is appended but stays silent
run set s "zz" n 5
run keys
run has s "zz"
run tracked s "zz"
run get s "zz"
run set s "zz" n 6
run dump

# --- both hooks see a stored value when it leaves undefined
run set s "difficulty" u
run set s "difficulty" n 3
run get s "difficulty"

# --- a pure dataset keeps plain, enumerable fields and never notifies
env hook s "gravity"
env pure b 1
run make
run keys
run tracked s "gravity"
run has s "gravity"
run has s "__is_world_dataset__"
run tracked s "__is_world_dataset__"
run set s "gravity" n 0.25
run set s "zz" n 1
run set s "gravity" n 0.25
run get s "gravity"

# --- the lazily built default instance is shared
env pure b 0
run default
run default
run get s "jump_height"
run set s "jump_height" n 12
run default
run get s "jump_height"
run tracked s "jump_height"
